cmake_minimum_required(VERSION 3.25)
project(cpp_ml_portfolio VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PORTFOLIO_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(PORTFOLIO_TORCH "Build the native CNN trainer" OFF)
option(PORTFOLIO_VISION "Build OpenCV image and camera applications" ON)
include(FetchContent)
include(cmake/native-deps.cmake)
include(cmake/runtime-local.cmake OPTIONAL)
foreach(dep eigen json httplib)
  string(TOUPPER ${dep} upper)
  FetchContent_Declare(${dep} URL ${${upper}_URL} URL_HASH SHA256=${${upper}_SHA256} DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR portfolio_no_build)
  FetchContent_MakeAvailable(${dep})
endforeach()
set(UTF8PROC_INSTALL OFF CACHE BOOL "" FORCE)
set(UTF8PROC_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(utf8proc URL ${UTF8PROC_URL} URL_HASH SHA256=${UTF8PROC_SHA256} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(utf8proc)
add_library(portfolio_options INTERFACE)
target_compile_options(portfolio_options INTERFACE $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-Wall;-Wextra;-Wpedantic>)
if(PORTFOLIO_SANITIZE)
  target_compile_options(portfolio_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(portfolio_options INTERFACE -fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(portfolio_http INTERFACE)
target_include_directories(portfolio_http SYSTEM INTERFACE ${httplib_SOURCE_DIR})
target_link_libraries(portfolio_http INTERFACE Threads::Threads)
if(OpenSSL_FOUND)
  target_compile_definitions(portfolio_http INTERFACE CPPHTTPLIB_OPENSSL_SUPPORT)
  target_link_libraries(portfolio_http INTERFACE OpenSSL::SSL OpenSSL::Crypto)
  if(APPLE)
    target_link_libraries(portfolio_http INTERFACE "-framework CoreFoundation" "-framework Security")
  endif()
endif()
add_library(portfolio_core src/data.cpp src/metrics.cpp src/tabular.cpp src/clustering.cpp src/isolation_forest.cpp src/stream.cpp src/recommendation.cpp src/registry.cpp src/tokenizer.cpp src/tracking.cpp)
target_include_directories(portfolio_core PUBLIC include)
target_include_directories(portfolio_core SYSTEM PUBLIC ${eigen_SOURCE_DIR} ${json_SOURCE_DIR}/single_include)
target_link_libraries(portfolio_core PUBLIC utf8proc portfolio_options portfolio_http OpenSSL::Crypto)
foreach(app ml_baselines clustering_explorer anomaly_monitor recommender experiment_tracker)
  add_executable(${app} apps/${app}.cpp)
  target_link_libraries(${app} PRIVATE portfolio_core)
endforeach()
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS ${ONNXRUNTIME_ROOT}/include)
find_library(ORT_LIBRARY NAMES onnxruntime HINTS ${ONNXRUNTIME_ROOT}/lib)
if(ORT_INCLUDE AND ORT_LIBRARY)
  add_library(portfolio_inference src/inference.cpp)
  target_include_directories(portfolio_inference SYSTEM PUBLIC ${ORT_INCLUDE})
  target_link_libraries(portfolio_inference PUBLIC portfolio_core ${ORT_LIBRARY})
  foreach(app inference_engine sentiment_service)
    add_executable(${app} apps/${app}.cpp)
    target_link_libraries(${app} PRIVATE portfolio_inference)
  endforeach()
  if(PORTFOLIO_VISION)
    find_package(OpenCV QUIET COMPONENTS core imgproc imgcodecs videoio highgui)
    if(OpenCV_FOUND)
      add_executable(vision_inspector apps/vision_inspector.cpp)
      target_include_directories(vision_inspector SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
      target_link_libraries(vision_inspector PRIVATE portfolio_inference ${OpenCV_LIBS})
    else()
      message(STATUS "OpenCV not found: vision_inspector is not built; install OpenCV and reconfigure.")
    endif()
  endif()
else()
  message(FATAL_ERROR "ONNX Runtime is required. Run python tools/bootstrap.py before configuring.")
endif()
if(PORTFOLIO_TORCH)
  find_package(Torch REQUIRED)
  add_executable(native_cnn apps/native_cnn.cpp)
  separate_arguments(TORCH_COMPILE_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
  target_compile_options(native_cnn PRIVATE ${TORCH_COMPILE_FLAGS})
  target_link_libraries(native_cnn PRIVATE portfolio_core ${TORCH_LIBRARIES})
endif()
include(CTest)
if(BUILD_TESTING)
  FetchContent_Declare(catch2 URL ${CATCH2_URL} URL_HASH SHA256=${CATCH2_SHA256} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(catch2)
  if(EXISTS "${CMAKE_SOURCE_DIR}/.venv/bin/python")
    set(Python3_EXECUTABLE "${CMAKE_SOURCE_DIR}/.venv/bin/python" CACHE FILEPATH "")
  endif()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/make_fixture.py ${CMAKE_BINARY_DIR}/test-model.onnx RESULT_VARIABLE fixture_result)
  if(NOT fixture_result EQUAL 0)
    message(FATAL_ERROR "Cannot generate ONNX test fixture. Install Python tools with uv sync.")
  endif()
  add_executable(portfolio_tests tests/test_data.cpp tests/test_algorithms.cpp tests/test_registry.cpp tests/test_runtime.cpp tests/test_tracking.cpp)
  target_compile_definitions(portfolio_tests PRIVATE PORTFOLIO_TEST_MODEL="${CMAKE_BINARY_DIR}/test-model.onnx")
  target_link_libraries(portfolio_tests PRIVATE portfolio_inference Catch2::Catch2WithMain)
  list(APPEND CMAKE_MODULE_PATH ${catch2_SOURCE_DIR}/extras)
  include(Catch)
  catch_discover_tests(portfolio_tests)
endif()
